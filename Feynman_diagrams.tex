\noindent
Consider the LSWT Hamiltonian $\mathcal H_2 = \sum_{kn} \epsilon_{kn} \alpha_{kn}^\dagger \alpha_{kn}$.
Suppose we have computed the following vertex factors (quantum matrix elements) for fixed momentum $\vec k$ and arbitrary momentum $\vec q \in \mathrm{BZ}$:
\begin{gather}
	g^{---}_{kq,nml} \alpha_{-k-q,n} \alpha_{qm} \alpha_{kl} \hspace{1.5cm}
	g^{+--}_{kq,nml} \alpha_{k+q,n}^\dagger \alpha_{q,m} \alpha_{kl} \hspace{1.5cm}
	g^{++-}_{kq,nml} \alpha_{k+q,n}^\dagger \alpha_{-q,m}^\dagger \alpha_{kl} \hspace{1.5cm}
	\\
	\tilde g^{-}_{n} \alpha_{0n} \hspace{1.5cm}
	\tilde h^{+-}_{k,nm} \alpha_{kn}^\dagger \alpha_{km}
\end{gather}
The first three vertex factors are due to the cubic vertex and are of order $\mathcal O(\sqrt S)$.
The fourth vertex is a commutator term coming from normal-ordering the cubic vertex and is of order $\mathcal O(\sqrt S)$.
The fifth vertex is a commutator term coming from normal-ordering the quartic vertex and is of order $\mathcal O(S^0)$. \\
All vertices are normal-ordered and symmetrized in their particle/hole degrees of freedom.

The Dyson equation reads
\begin{align}
	G = G^0 + G^0 \Sigma G
\end{align}
with the non-interacting propagator $G^0$, the interacting propagator $G$ and the self-energy $\Sigma$.
Diagramatically, we assign a negative propagator to every line due to the sign the propagator $G(\tau) \sim -\braket{T_\tau a(\tau) a^\dagger(0)}$ is defined with. We obtain: \\
\includegraphics{Dyson_equation.png} \\
So the one-particle irreducible diagrams we draw acually give us the \emph{negative} of the self-energy. \\
\\
The Feynman rules for the sign of a particular diagram are given as follows:
\begin{itemize}
	\item a global $-1$ due to the above argument,
	\item a $-1$ for every internal propagator,
	\item a $(-1)^n$ for a diagram of order $n$ (i.e., with $n$ vertices); this factor is due to the expansion of the $e^{-S_\mathrm{int}}$ factor in the path integral.
\end{itemize}
	
\section{One-Loop}
	
\noindent
Compute exact expressions for all one-loop Feynman diagrams.
We will need the bosonic Matsubara sum
\begin{align}
	\frac{1}{\beta} \sum_{i\omega} \frac{1}{(i\omega - \xi_1) (i\omega - \xi_2)} &= -\frac{n_B(\xi_1) - n_B(\xi_2)}{\xi_1 - \xi_2} \\
	\Rightarrow \frac{1}{\beta} \sum_{i\omega} \frac{1}{(\eta_1 i\omega - \xi_1) (\eta_2 (i\Omega - i\omega) - \xi_2)} &= -\eta_1 \eta_2 \frac{n_B(\eta_1 \xi_1) - n_B(-\eta_2 \xi_2)}{i\Omega - \eta_1 \xi_1 - \eta_2 \xi_2} \qquad (*)
\end{align}
with $\eta_1, \eta_2 \in \{+, -\}$, where we have used that $n_B(i\Omega + z) = n_B(z)$ for bosonic Matsubara frequencies $i\Omega$. \\
We will also use that $n_B(-z) = -1 - n_B(z)$.


\subsection{Particle-Particle Bubble}

\includegraphics{Feynman_p,pp,p_bubble.png} \\
By the Feynman rules, we find the following expression:
\begin{align}
	\Sigma_{nm}^\mathrm{p,pp,p-bubble}(i\Omega, \vec k) &= -(-1)^2 (-1)^2 \frac{2}{\beta N} \sum_{\vec q,ll'} g^{++-}_{kq,ll'n} (g^{++-}_{kq,ll'm})^* \sum_{i\omega} \frac{1}{(i\omega - \epsilon_{-q,l}) (i\Omega - i\omega - \epsilon_{k+q,l'})} \\
%
	&= \frac{2}{N} \sum_{\vec q,ll'} g^{++-}_{kq,ll'n} (g^{++-}_{kq,ll'm})^* \, \frac{n_B(\epsilon_{-q,l}) - n_B(-\epsilon_{k+q,l'})}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}} \\
	%
	&= \frac{2}{N} \sum_{\vec q,ll'} g^{++-}_{kq,ll'n} (g^{++-}_{kq,ll'm})^* \, \frac{1 + n_B(\epsilon_{-q,l}) + n_B(\epsilon_{k+q,l'})}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}} \\
%
	&\overset{T\to 0}{\longrightarrow} \frac{2}{N} \sum_{\vec q,ll'} \frac{g^{++-}_{kq,ll'n} (g^{++-}_{kq,ll'm})^*}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}}
\end{align}
The symmetry factor $2$ comes about because there are two possible ways to wire the loop.

\noindent
\includegraphics{Feynman_p,pp,h_bubble.png} \\
By the Feynman rules, we find the following expression:
\begin{align}
	\Sigma_{nm}^\mathrm{p,pp,h-bubble}(i\Omega, \vec k) &= -(-1)^2 (-1)^2 \frac{2 \cdot 3}{\beta N} \sum_{\vec q,ll'} g^{++-}_{kq,ll'n} (g^{+++}_{kq,ll'm})^* \sum_{i\omega} \frac{1}{(i\omega - \epsilon_{-q,l}) (i\Omega - i\omega - \epsilon_{k+q,l'})} \\
	%
	&= \frac{6}{N} \sum_{\vec q,ll'} g^{++-}_{kq,ll'n} (g^{+++}_{kq,ll'm})^* \, \frac{n_B(\epsilon_{-q,l}) - n_B(-\epsilon_{k+q,l'})}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}} \\
	%
	&= \frac{6}{N} \sum_{\vec q,ll'} g^{++-}_{kq,ll'n} (g^{+++}_{kq,ll'm})^* \, \frac{1 + n_B(\epsilon_{-q,l}) + n_B(\epsilon_{k+q,l'})}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}} \\
	%
	&\overset{T\to 0}{\longrightarrow} \frac{6}{N} \sum_{\vec q,ll'} \frac{g^{++-}_{kq,ll'n} (g^{+++}_{kq,ll'm})^*}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}}
\end{align}
The symmetry factor is $2 \cdot 3$ because there are three possible ways to wire the right vertex to the outgoing external leg, and two possible ways to wire the loop.

\noindent

	\Sigma_{nm}^\mathrm{h,pp,p-bubble}(i\Omega, \vec k) &= -(-1)^2 (-1)^2 \frac{2 \cdot 3}{\beta N} \sum_{\vec q,ll'} g^{+++}_{kq,ll'n} (g^{++-}_{kq,ll'm})^* \sum_{i\omega} \frac{1}{(i\omega - \epsilon_{-q,l}) (i\Omega - i\omega - \epsilon_{k+q,l'})} \\
	%
	&= \frac{6}{N} \sum_{\vec q,ll'} g^{+++}_{kq,ll'n} (g^{++-}_{kq,ll'm})^* \, \frac{n_B(\epsilon_{-q,l}) - n_B(-\epsilon_{k+q,l'})}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}} \\
	%
	&= \frac{6}{N} \sum_{\vec q,ll'} g^{+++}_{kq,ll'n} (g^{++-}_{kq,ll'm})^* \, \frac{1 + n_B(\epsilon_{-q,l}) + n_B(\epsilon_{k+q,l'})}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}} \\
	%
	&\overset{T\to 0}{\longrightarrow} \frac{6}{N} \sum_{\vec q,ll'} \frac{g^{+++}_{kq,ll'n} (g^{++-}_{kq,ll'm})^*}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}}
\end{align}
The symmetry factor is $2 \cdot 3$ because there are three possible ways to wire the left vertex to the incoming external leg, and two possible ways to wire the loop.

\noindent
\includegraphics{Feynman_h,pp,h_bubble.png} \\
By the Feynman rules, we find the following expression:
\begin{align}
	\Sigma_{nm}^\mathrm{h,pp,h-bubble}(i\Omega, \vec k) &= -(-1)^2 (-1)^2 \frac{2 \cdot 3 \cdot 3}{\beta N} \sum_{\vec q,ll'} g^{+++}_{kq,ll'n} (g^{+++}_{kq,ll'm})^* \sum_{i\omega} \frac{1}{(i\omega - \epsilon_{-q,l}) (i\Omega - i\omega - \epsilon_{k+q,l'})} \\

	%
	&= \frac{18}{N} \sum_{\vec q,ll'} g^{+++}_{kq,ll'n} (g^{+++}_{kq,ll'm})^* \, \frac{1 + n_B(\epsilon_{-q,l}) + n_B(\epsilon_{k+q,l'})}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}} \\
	%
	&\overset{T\to 0}{\longrightarrow} \frac{18}{N} \sum_{\vec q,ll'} \frac{g^{+++}_{kq,ll'n} (g^{+++}_{kq,ll'm})^*}{i\Omega - \epsilon_{-q,l} - \epsilon_{k+q,l'}}
\end{align}

\subsection{Particle-Hole Bubble}
\includegraphics{Feynman_ph_bubble.png} \\
By the Feynman rules, we find the following expression:
\begin{align}
	\Sigma_{nm}^\mathrm{p,ph,p-bubble}(i\Omega, \vec k) &= -(-1)^2 (-1)^2 \frac{2\cdot 2}{\beta N} \sum_{\vec q,ll'} g^{+--}_{kq,ll'n} (g^{+--}_{kq,ll'm})^* \sum_{i\omega} \frac{1}{(i\omega - \epsilon_{q,l}) (-i\Omega + i\omega - \epsilon_{k+q,l'})} \\
	%
	&\overset{(*)}{=} -\frac{4}{N} \sum_{\vec q,ll'} g^{+--}_{kq,ll'n} (g^{+--}_{kq,ll'm})^* \, \frac{n_B(\epsilon_{q,l}) - n_B(\epsilon_{k+q,l'})}{i\Omega - \epsilon_{q,l} + \epsilon_{k+q,l'})} \\
	%
	&= -\frac{4}{N} \sum_{\vec q,ll'} g^{+--}_{kq,ll'n} (g^{+--}_{kq,ll'm})^* \, \frac{n_B(\epsilon_{q,l}) - n_B(\epsilon_{k+q,l'})}{i\Omega - \epsilon_{q,l} + \epsilon_{k+q,l'}} \\
	%
	&\overset{T\to 0}{\longrightarrow} 0
\end{align}
The symmetry factor is $2\cdot 2 = 4$ because there are two ways to wire each vertex to an external leg, and one way to wire the loop. \\
All other combinations of in- and outgoing particles/holes also yield zero.

\subsection{Hole-Hole Bubble}
\includegraphics{Feynman_p,hh,p_bubble.png} \\
By the Feynman rules, we find the following expression:
\begin{align}
	\Sigma_{nm}^\mathrm{p,hh,p-bubble}(i\Omega, \vec k) &= -(-1)^2 (-1)^2 \frac{3\cdot 3\cdot 2}{\beta N} \sum_{\vec q,ll'} g^{---}_{kq,ll'n} (g^{---}_{kq,ll'm})^* \sum_{i\omega} \frac{1}{(-i\omega - \epsilon_{q,l}) (-i\Omega + i\omega - \epsilon_{-k-q,l'})} \\
	%
	&\overset{(*)}{=} \frac{18}{N} \sum_{\vec q,ll'} g^{---}_{kq,ll'n} (g^{---}_{kq,ll'm})^* \, \frac{n_B(-\epsilon_{q,l}) - n_B(\epsilon_{-k-q,l'})}{i\Omega + \epsilon_{q,l} + \epsilon_{-k-q,l'}} \\
	%
	&= -\frac{18}{N} \sum_{\vec q,ll'} g^{---}_{kq,ll'n} (g^{---}_{kq,ll'm})^* \, \frac{1 + n_B(\epsilon_{q,l}) + n_B(\epsilon_{-k-q,l'})}{i\Omega + \epsilon_{q,l} + \epsilon_{-k-q,l'}} \\
	%
	&\overset{T\to 0}{\longrightarrow} -\frac{18}{N} \sum_{\vec q,ll'} \frac{g^{---}_{kq,ll'n} (g^{---}_{kq,ll'm})^*}{i\Omega + \epsilon_{q,l} + \epsilon_{-k-q,l'}}
\end{align}
The symmetry factor $3\cdot 3 = 18$ comes about because there are three ways to wire each vertex to the incoming external leg, and two ways to wire the loop.

\subsection{Particle Stub (``Particle Tadpole")}
\includegraphics{Feynman_p,p,p_stub.png} \\
By the Feynman rules, we find the following expression:
\begin{align}
	\Sigma_{nm}^\mathrm{p,p,p-stub}(\vec k) &= -(-1) (-1)^2 \frac{2}{N} \sum_{l} \frac{g^{++-}_{k0,mln} \tilde g^{-}_{l}}{-\epsilon_{0l}}
%
	= -\frac{2}{N} \sum_{l} \frac{g^{++-}_{k0,mln} \tilde g^{-}_{l}}{\epsilon_{0l}}
\end{align}
The symmetry factor is $2$ because there is one possible way to wire the cubic vertex to the incoming external leg, two ways to wire the cubic vertex to the outgoing external leg, and one remaining way to wire the two vertices together. \\
Note that the diagram is frequency-independent.
