We need
\begin{align}
	\mathcal S &= \left(\mathcal U_{2n,2t}^{-1}\right)^{-1} \left(\mathcal U_{2n,2t+1}\right)
\end{align}
Create single magnon excitation at position $(i, s)$:
\begin{align}
	a_{is}^\dagger \ket{\mathrm{GS}} &= \frac{1}{\sqrt N} \sum_{kn} e^{ikr_i} (\mathcal U_{2s+1,2n}(-k) \alpha_{-kn} + \mathcal U_{2s+1,2n+1}(-k) \alpha_{kn}^\dagger) \ket{\mathrm{GS}} \\
%
	&= \frac{1}{\sqrt N} \sum_{kn} e^{ikr_i} \mathcal U_{2s+1,2n+1}(-k) \alpha_{kn}^\dagger \ket{\mathrm{GS}}
\end{align}
Hamiltonian: $\hat H = \sum_{qm} \epsilon_{qm} \alpha_{qm}^\dagger \alpha_{qm}$.
Time evolution of wave packet:
\begin{align}
	\sum_{is} \psi_s(r_i) a_{is}^\dagger \ket{\mathrm{GS}} &= e^{-i\hat H t} \frac{1}{\sqrt N} \sum_{is} \sum_{kn} \psi_s(r_i) e^{ikr_i} \mathcal U_{2s+1,2n+1}(-k) \alpha_{kn}^\dagger \ket{\mathrm{GS}} \\
%
	&= \frac{1}{\sqrt N} \sum_{qm} \sum_{is} \sum_{kn}\psi_s(r_i) e^{ikr_i - i\epsilon_{qm} t} \mathcal U_{2s+1,2n+1}(-k) \alpha_{qm}^\dagger \alpha_{qm} \alpha_{kn}^\dagger \ket{\mathrm{GS}} \\
%
	&= \frac{1}{\sqrt N} \sum_{is} \psi_s(r_i) \sum_{kn} e^{ikr_i - i\epsilon_{kn} t} \mathcal U_{2s+1,2n+1}(-k) \alpha_{kn}^\dagger \ket{\mathrm{GS}} \\
%
	&\equiv \sum_{js'} \underbrace{ \frac{1}{N} \sum_{is} \psi_s(r_i) \sum_{kn} e^{ik(r_i - r_j) - i\epsilon_{kn} t} \mathcal U_{2s+1,2n+1}(-k) \mathcal U_{2n+1,2s'+1}^{-1}(-k) }_{\equiv \psi_{s'}(r_j, t)} a_{js'}^\dagger \ket{\mathrm{GS}} \\
%
	&= \sum_{js'} \psi_{s'}(r_j, t) a_{js'}^\dagger \ket{\mathrm{GS}}
\end{align}
Alternative using infinitesimal time slices:
\begin{align}
	\sum_{is} \psi_s(r_i, t + \delta t) a_{is}^\dagger \ket{\mathrm{GS}} &= e^{-i\hat H \delta t} \sum_{is} \psi_s(r_i, t) a_{is}^\dagger \ket{\mathrm{GS}} \\
%
	&= \sum_{is} \psi_s(r_i, t) (\mathbbm 1 - i\hat H \delta t) a_{is}^\dagger \ket{\mathrm{GS}} \\
%
	&= \sum_{is} \psi_s(r_i, t) a_{is}^\dagger \ket{\mathrm{GS}} - i\delta t\sum_{is} \psi_s(r_i, t) \sum_j h_{ij} 
\end{align}
